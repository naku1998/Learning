///StoRE PROCEDURE WITH OUTput PARAMETERS means procedure 
is returning a value like a function and it always return an integer value

create procedure _COUNT
@a int output                     // @a variable is returning value of (count (ID) from Sample )
as begin
select @a=count (ID) from Sample 
end

declare @b int             // we also need to save output in a variable @b is that variable and we are declring it
execute _COUNT  @a= @b out       //puting value of @a in @b
print @b
