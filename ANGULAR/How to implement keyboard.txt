-------------------.ts----------------

 capsLock:boolean= true;
  clearall:boolean;
  selectfield:any;
  selectedfield(id:any)
  {
   this.selectfield = id;
    this.clearall = true;
  }

  number: string = "";
  press(event: any) {
    if(this.clearall){
      this.number = "";
      this.clearall= false;
    }
    if (event.target.value != undefined && event.target.value != "") {
      this.number = this.number + event.target.value;
      this.add();
    }
    else {
      this.number = this.number.slice(0, -1);
     this.add();
    }
  }

  add()
  {
    if(this.selectfield == 1)
    {
      this.showaxisedit.displayName = this.number;
    }
    if(this.selectfield == 2)
    {
      this.showaxisedit.actualPosition = this.number;
    }
    else if(this.selectfield == 3 )
    {
      this.showaxisedit.maxLoad = this.number;
    }
    else if(this.selectfield == 4)
    {
      this.showaxisedit.loadCellWarning = this.number;
    }
    else if(this.selectfield == 5 )
    {
      this.showaxisedit.loadCellLimit = this.number;
    }
    else if(this.selectfield == 6 )
    {
      this.showaxisedit.loadCellMeasuringDelay = this.number;
    }
    else if(this.selectfield == 7 )
    {
      this.showaxisedit.softDownLimit = this.number;
    }
    else if(this.selectfield == 8)
    {
      this.showaxisedit.softUpLimit = this.number;
    }
    else if(this.selectfield == 9)
    {
      this.showaxisedit.riggedSoftDownLimit = this.number;
    }
    else if(this.selectfield == 10)
    {
      this.showaxisedit.riggedSoftUpLimit = this.number;
    }
    else if(this.selectfield == 11)
    {
      this.showaxisedit.minVelocity = this.number;
    }
    else if(this.selectfield == 12)
    {
      this.showaxisedit.defaultAcceleration= this.number;
    }
    else if(this.selectfield == 13)
    {
      this.showaxisedit.maxAcceleration = this.number;
    }
    else if(this.selectfield == 14)
    {
      this.showaxisedit.defaultVelocity = this.number;
    }
    else if(this.selectfield == 15)
    {
      this.showaxisedit.maxVelocity = this.number;
    }
    else if(this.selectfield == 16)
    {
      this.showaxisedit.defaultDeceleration = this.number;
    }
    else if(this.selectfield == 17 )
    {
      this.showaxisedit.maxDeceleration = this.number;
    }
  }
